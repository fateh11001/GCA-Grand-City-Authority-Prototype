function addBox(
    x, y, z,
    width, height, depth,
    material,
    collision = false,
    castShadow = false
) {
    const mesh = new THREE.Mesh(unitBoxGeometry, material);

    mesh.position.set(x, y, z);
    mesh.scale.set(width, height, depth);

    mesh.castShadow = castShadow;
    mesh.receiveShadow = false;

    scene.add(mesh);

    if (collision) {
        addCollider(x, z, width, depth);
    }

    return mesh;
}

function addCylinder(
    x, y, z,
    radiusTop, radiusBottom, height,
    material,
    collision = false
) {
    const geometry = new THREE.CylinderGeometry(
        radiusTop,
        radiusBottom,
        height,
        10
    );

    const mesh = new THREE.Mesh(geometry, material);

    mesh.position.set(x, y, z);
    mesh.castShadow = false;
    mesh.receiveShadow = false;

    scene.add(mesh);

    if (collision) {
        const radius = Math.max(radiusTop, radiusBottom);
        addCollider(x, z, radius * 2, radius * 2);
    }

    return mesh;
}

const windowPositions = [];
const framePositions = [];

const windowGeometry = new THREE.BoxGeometry(1.45, 1.9, 0.12);
const frameGeometry = new THREE.BoxGeometry(1.55, 0.07, 0.16);

const dummy = new THREE.Object3D();

function queueWindow(x, y, z, rotationY = 0) {
    windowPositions.push({ x, y, z, rotationY });

    framePositions.push({
        x, y: y - 1, z, rotationY
    });

    framePositions.push({
        x, y: y + 1, z, rotationY
    });
}

function buildWindowInstances() {
    if (!windowPositions.length) return;

    const windows = new THREE.InstancedMesh(
        windowGeometry,
        mat.glass,
        windowPositions.length
    );

    windows.castShadow = false;
    windows.receiveShadow = false;

    windowPositions.forEach((p, i) => {
        dummy.position.set(p.x, p.y, p.z);
        dummy.rotation.set(0, p.rotationY, 0);
        dummy.scale.set(1, 1, 1);
        dummy.updateMatrix();

        windows.setMatrixAt(i, dummy.matrix);
    });

    windows.instanceMatrix.needsUpdate = true;
    scene.add(windows);

    const frames = new THREE.InstancedMesh(
        frameGeometry,
        mat.concrete,
        framePositions.length
    );

    frames.castShadow = false;
    frames.receiveShadow = false;

    framePositions.forEach((p, i) => {
        dummy.position.set(p.x, p.y, p.z);
        dummy.rotation.set(0, p.rotationY, 0);
        dummy.scale.set(1, 1, 1);
        dummy.updateMatrix();

        frames.setMatrixAt(i, dummy.matrix);
    });

    frames.instanceMatrix.needsUpdate = true;
    scene.add(frames);
}

        maxZ: z + depth / 2 + 1

function isTreeSpotClear(x, z) {
    const roadHalfWidth = 9;
    const treeRoadClearance = 4.5;

    const nearestCrossStreet =
        -210 + Math.round((z + 210) / 60) * 60;

    if (
        Math.abs(z - nearestCrossStreet) <
        roadHalfWidth + treeRoadClearance
    ) {
        return false;
    }

    const buildingClearance = 3.0;

    const insideBuilding = buildingFootprints.some((building) =>
        x > building.minX - buildingClearance &&
        x < building.maxX + buildingClearance &&
        z > building.minZ - buildingClearance &&
        z < building.maxZ + buildingClearance
    );

    if (insideBuilding) return false;

    return treePositionList.every(([tx, tz]) =>
        Math.hypot(x - tx, z - tz) >= 13
    );
}

function prepareInteractiveTree(gltfScene) {
    gltfScene.updateMatrixWorld(true);

    const bounds = new THREE.Box3().setFromObject(gltfScene);
    const size = new THREE.Vector3();

    bounds.getSize(size);

    if (!Number.isFinite(size.y) || size.y <= 0) {
        throw new Error("Invalid Jacaranda model dimensions.");
    }

    const scale = SETTINGS.treeHeight / size.y;

    const offset = new THREE.Vector3(
        -(bounds.min.x + bounds.max.x) * 0.5 * scale,
        -bounds.min.y * scale,
        -(bounds.min.z + bounds.max.z) * 0.5 * scale
    );

    const root = new THREE.Group();
    const chunkSize = SETTINGS.treeChunkSize;

    function buildChunks(source, group, material, type) {
        const sourceGeometry = source.geometry;
        const sourcePosition = sourceGeometry.getAttribute("position");
        const sourceNormal = sourceGeometry.getAttribute("normal");
        const sourceUV = sourceGeometry.getAttribute("uv");
        const sourceIndex = sourceGeometry.index;

        if (!sourcePosition) return;

        const normalMatrix = new THREE.Matrix3()
            .getNormalMatrix(source.matrixWorld);

        const chunks = new Map();

        const start = group.start;
        const count = group.count;

        const vertexPosition = new THREE.Vector3();
        const vertexNormal = new THREE.Vector3();

        for (let i = start; i + 2 < start + count; i += 3) {
            const triangle = [];
            const centroid = new THREE.Vector3();

            for (let j = 0; j < 3; j++) {
                const id = sourceIndex
                    ? sourceIndex.getX(i + j)
                    : i + j;

                vertexPosition
                    .fromBufferAttribute(sourcePosition, id)
                    .applyMatrix4(source.matrixWorld)
                    .multiplyScalar(scale)
                    .add(offset);

                const p = vertexPosition.clone();

                if (sourceNormal) {
                    vertexNormal
                        .fromBufferAttribute(sourceNormal, id)
                        .applyMatrix3(normalMatrix)
                        .normalize();
                } else {
                    vertexNormal.set(0, 1, 0);
                }

                const n = vertexNormal.clone();

                const uv = sourceUV
                    ? new THREE.Vector2().fromBufferAttribute(sourceUV, id)
                    : new THREE.Vector2();

                triangle.push({ p, n, uv });
                centroid.add(p);
            }

            centroid.multiplyScalar(1 / 3);

            const key = [
                Math.floor(centroid.x / chunkSize),
                Math.floor(centroid.y / chunkSize),
                Math.floor(centroid.z / chunkSize)
            ].join("_");

            if (!chunks.has(key)) {
                chunks.set(key, {
                    vertices: [],
                    min: new THREE.Vector3(Infinity, Infinity, Infinity),
                    max: new THREE.Vector3(-Infinity, -Infinity, -Infinity)
                });
            }

            const chunk = chunks.get(key);

            for (const vertex of triangle) {
                chunk.vertices.push(vertex);
                chunk.min.min(vertex.p);
                chunk.max.max(vertex.p);
            }
        }

        for (const chunk of chunks.values()) {
            const center = chunk.min.clone()
                .add(chunk.max)
                .multiplyScalar(0.5);

            const positions = [];
            const normals = [];
            const uvs = [];

            let radius = 0;

            for (const vertex of chunk.vertices) {
                const local = vertex.p.clone().sub(center);

                positions.push(local.x, local.y, local.z);
                normals.push(vertex.n.x, vertex.n.y, vertex.n.z);
                uvs.push(vertex.uv.x, vertex.uv.y);

                radius = Math.max(
                    radius,
                    Math.hypot(local.x, local.z)
                );
            }

            const geometry = new THREE.BufferGeometry();

            geometry.setAttribute(
                "position",
                new THREE.Float32BufferAttribute(positions, 3)
            );

            geometry.setAttribute(
                "normal",
                new THREE.Float32BufferAttribute(normals, 3)
            );

            if (sourceUV) {
                geometry.setAttribute(
                    "uv",
                    new THREE.Float32BufferAttribute(uvs, 2)
                );
            }

            geometry.computeBoundingSphere();

            const pivot = new THREE.Group();

            const anchor = new THREE.Vector3(
                center.x,
                chunk.min.y,
                center.z
            );

            pivot.position.copy(anchor);

            const mesh = new THREE.Mesh(geometry, material);
            mesh.position.copy(center).sub(anchor);

            mesh.castShadow = false;
            mesh.receiveShadow = false;

            pivot.add(mesh);
            root.add(pivot);

            pivot.userData.gcaTreePart = {
                type,
                center: center.clone(),
                anchor: anchor.clone(),
                radius: Math.max(radius, 0.2),
                minY: chunk.min.y,
                maxY: chunk.max.y
            };
        }
    }

    gltfScene.traverse((source) => {
        if (!source.isMesh) return;

        const materials = Array.isArray(source.material)
            ? source.material
            : [source.material];

        const groups = source.geometry.groups.length
            ? source.geometry.groups
            : [{
                start: 0,
                count: source.geometry.index
                    ? source.geometry.index.count
                    : source.geometry.getAttribute("position").count,
                materialIndex: 0
            }];

        for (const group of groups) {
            const material = materials[group.materialIndex];

            if (!material) continue;

            const name =
                `${source.name || ""} ${material.name || ""}`.toLowerCase();

            let type = "branch";

            if (/leaf|leaves|foliage|canopy/.test(name)) {
                type = "leaf";
            } else if (/trunk|bark|stem|wood/.test(name)) {
                type = "trunk";
            }

            buildChunks(source, group, material, type);
        }
    });

    return { root };
}

function collectTreeParts(root) {
    const parts = [];

    root.traverse((object) => {
        const data = object.userData.gcaTreePart;

        if (!data) return;

        parts.push({
            pivot: object,
            type: data.type,
            center: new THREE.Vector3(
                data.center.x,
                data.center.y,
                data.center.z
            ),
            radius: data.radius,
            minY: data.minY,
            maxY: data.maxY,
            bendX: 0,
            bendZ: 0
        });
    });

    return parts;
}

function pointInsideBuilding(x, z, padding = 0) {
    return buildingFootprints.some((b) =>
        x > b.minX - padding &&
        x < b.maxX + padding &&
        z > b.minZ - padding &&
        z < b.maxZ + padding
    );
}

function pointNearTree(x, z, padding = 0) {
    return treePositionList.some(([tx, tz]) =>
        Math.hypot(x - tx, z - tz) < 4.5 + padding
    );
}

// Prevent benches and bins from being placed on top of one another.

const propReservations = [];

function reservePropSpot(x, z, radius) {
    if (pointInsideBuilding(x, z, 2)) return false;
    if (pointNearTree(x, z, 2)) return false;

    const overlapsAnotherProp = propReservations.some((prop) =>
        Math.hypot(x - prop.x, z - prop.z) <
        radius + prop.radius
    );

    if (overlapsAnotherProp) return false;

    propReservations.push({ x, z, radius });

    return true;
}

function collidesAt(x, z) {
    const radius = SETTINGS.playerRadius;

    const worldLimit = SETTINGS.worldHalfSize - radius;

    if (
        x < -worldLimit ||
        x > worldLimit ||
        z < -worldLimit ||
        z > worldLimit
    ) {
        return true;
    }

    for (const box of colliders) {
        if (
            x + radius > box.minX &&
            x - radius < box.maxX &&
            z + radius > box.minZ &&
            z - radius < box.maxZ
        ) {
            return true;
        }
    }

    return false;
}

// TREE COLLISION
// ============================================================

const playerBodyCenter = new THREE.Vector3();
const treePartWorldCenter = new THREE.Vector3();
const awayDirection = new THREE.Vector3();
const inverseTreeRotation = new THREE.Quaternion();

function treeCollisionAt(x, z, playerY) {
    for (const tree of interactiveTrees) {
        const dx = x - tree.trunkCollider.x;
        const dz = z - tree.trunkCollider.z;

        const distanceSquared = dx * dx + dz * dz;

        const collisionRadius =
            SETTINGS.playerRadius + tree.trunkCollider.radius;

        if (
            distanceSquared <
            collisionRadius * collisionRadius
        ) {
            return true;
        }
    }

    return false;
}

function updateTreeInteractions(delta) {
    playerBodyCenter.set(
        player.position.x,
        player.position.y + 1.35,
        player.position.z
    );

    for (const tree of interactiveTrees) {
        inverseTreeRotation
            .copy(tree.root.quaternion)
            .invert();

        for (const part of tree.parts) {
            let targetX = 0;
            let targetZ = 0;

            if (part.type !== "trunk") {
                treePartWorldCenter
                    .copy(part.center)
                    .applyQuaternion(tree.root.quaternion)
                    .add(tree.root.position);

                const dx =
                    playerBodyCenter.x - treePartWorldCenter.x;

                const dy =
                    playerBodyCenter.y - treePartWorldCenter.y;

                const dz =
                    playerBodyCenter.z - treePartWorldCenter.z;

                const distance = Math.sqrt(
                    dx * dx + dy * dy + dz * dz
                );

                const contactRadius =
                    part.radius +
                    SETTINGS.playerRadius +
                    SETTINGS.treeCollisionPadding;

                if (distance < contactRadius) {
                    awayDirection.set(-dx, 0, -dz);

                    if (awayDirection.lengthSq() > 0.0001) {
                        awayDirection.normalize();
                        awayDirection.applyQuaternion(inverseTreeRotation);

                        const strength =
                            1 - distance / contactRadius;

                        const amount =
                            strength * SETTINGS.treeLeafPushStrength;

                        targetX = awayDirection.z * amount;
                        targetZ = -awayDirection.x * amount;

                        if (part.type === "branch") {
                            const ratio =
                                SETTINGS.treeBranchPushStrength /
                                SETTINGS.treeLeafPushStrength;

                            targetX *= ratio;
                            targetZ *= ratio;
                        }
                    }
                }
            }

            const recovery = part.type === "branch"
                ? SETTINGS.treeBranchRecovery
                : SETTINGS.treeLeafRecovery;

            const blend = 1 - Math.exp(-delta * recovery);

            part.bendX += (targetX - part.bendX) * blend;
            part.bendZ += (targetZ - part.bendZ) * blend;

            part.pivot.rotation.x = part.bendX;
            part.pivot.rotation.z = part.bendZ;
        }
    }
}

function updateCamera() {
    const distance = 7;
    const height = 3.1;

    const horizontalDistance =
        distance * Math.cos(cameraPitch);

    cameraTarget.set(
        player.position.x,
        player.position.y + 1.5,
        player.position.z
    );

    cameraDesired.set(
        player.position.x +
            Math.sin(cameraYaw) * horizontalDistance,

        player.position.y +
            height +
            Math.sin(cameraPitch) * distance,

        player.position.z +
            Math.cos(cameraYaw) * horizontalDistance
    );

    cameraDirection
        .copy(cameraDesired)
        .sub(cameraTarget);

    const desiredDistance = cameraDirection.length();

    cameraDirection.normalize();

    cameraRaycaster.set(cameraTarget, cameraDirection);
    cameraRaycaster.far = desiredDistance;

    const hits = cameraRaycaster.intersectObjects(
        cameraBlockers,
        false
    );

    let allowedDistance = desiredDistance;

    if (hits.length) {
        allowedDistance = Math.max(
            0.7,
            hits[0].distance - 0.35
        );
    }

    camera.position
        .copy(cameraTarget)
        .addScaledVector(cameraDirection, allowedDistance);

    camera.lookAt(cameraTarget);
}

// The animation uses connected pivot groups.
// The arms swing opposite to the legs.
// Knees bend during the backward part of the step.

let walkCycle = 0;
let animationBlend = 0;

const animationState = {
    legSwing: 0,
    kneeBend: 0,
    armSwing: 0,
    elbowBend: 0,
    bodyBob: 0,
    bodyLean: 0
};

function smoothTowards(current, target, speed, delta) {
    return THREE.MathUtils.damp(
        current,
        target,
        speed,
        delta
    );
}

function updateCharacterAnimation(delta, moving, running) {
    const targetBlend = moving ? 1 : 0;

    animationBlend = smoothTowards(
        animationBlend,
        targetBlend,
        moving ? 12 : 9,
        delta
    );

    if (moving) {
        walkCycle += delta * (running ? 13 : 9);
    }

    const stride = Math.sin(walkCycle);
    const oppositeStride = Math.sin(walkCycle + Math.PI);

    const legTarget = stride * 0.48;
    const kneeTarget = Math.max(0, -stride) * 0.55;

    const armTarget = -stride * 0.38;
    const elbowTarget = Math.max(0, stride) * 0.25;

    animationState.legSwing = smoothTowards(
        animationState.legSwing,
        legTarget * animationBlend,
        14,
        delta
    );

    animationState.kneeBend = smoothTowards(
        animationState.kneeBend,
        kneeTarget * animationBlend,
        14,
        delta
    );

    animationState.armSwing = smoothTowards(
        animationState.armSwing,
        armTarget * animationBlend,
        14,
        delta
    );

    animationState.elbowBend = smoothTowards(
        animationState.elbowBend,
        elbowTarget * animationBlend,
        14,
        delta
    );

    const bobTarget =
        moving ? Math.abs(Math.sin(walkCycle * 2)) * 0.055 : 0;

    animationState.bodyBob = smoothTowards(
        animationState.bodyBob,
        bobTarget,
        12,
        delta
    );

    const leanTarget = moving ? 0.055 : 0;

    animationState.bodyLean = smoothTowards(
        animationState.bodyLean,
        leanTarget,
        8,
        delta
    );

    // Legs: opposite movement, with a small knee bend.

    for (const leg of legs) {
        const phase = leg.side === 1 ? 0 : Math.PI;

        leg.hip.rotation.x =
            Math.sin(walkCycle + phase) *
            0.48 *
            animationBlend;

        leg.knee.rotation.x =
            Math.max(
                0,
                -Math.sin(walkCycle + phase)
            ) *
            0.55 *
            animationBlend;
    }

    // Arms: opposite to the legs.

    for (const arm of arms) {
        const phase = arm.side === 1 ? Math.PI : 0;

        arm.shoulder.rotation.x =
            Math.sin(walkCycle + phase) *
            0.38 *
            animationBlend;

        arm.elbow.rotation.x =
            Math.max(
                0,
                Math.sin(walkCycle + phase)
            ) *
            0.25 *
            animationBlend;
    }

    torso.position.y = 1.65 + animationState.bodyBob;
    torso.rotation.x = -animationState.bodyLean;

    head.rotation.x = animationState.bodyLean * 0.35;
}

const clock = new THREE.Clock();

function updatePlayer(delta) {
    let moveX = 0;
    let moveZ = 0;

    if (keys.KeyW || keys.ArrowUp) moveZ -= 1;
    if (keys.KeyS || keys.ArrowDown) moveZ += 1;
    if (keys.KeyA || keys.ArrowLeft) moveX -= 1;
    if (keys.KeyD || keys.ArrowRight) moveX += 1;

    const moving = moveX !== 0 || moveZ !== 0;

    const running =
        keys.ShiftLeft || keys.ShiftRight;

        const length = Math.hypot(moveX, moveZ);

        moveX /= length;
        moveZ /= length;

        const sin = Math.sin(cameraYaw);
        const cos = Math.cos(cameraYaw);

        const worldX = moveX * cos + moveZ * sin;
        const worldZ = -moveX * sin + moveZ * cos;

        const speed = running
            ? SETTINGS.runSpeed
            : SETTINGS.walkSpeed;

        const nextX =
            player.position.x + worldX * speed * delta;

        const nextZ =
            player.position.z + worldZ * speed * delta;

        if (
            !collidesAt(nextX, player.position.z) &&
            !treeCollisionAt(nextX, player.position.z, player.position.y)
        ) {
            player.position.x = nextX;
        }

        if (
            !collidesAt(player.position.x, nextZ) &&
            !treeCollisionAt(player.position.x, nextZ, player.position.y)
        ) {
            player.position.z = nextZ;
        }

        player.lookAt(
            player.position.x + worldX,
            player.position.y,
            player.position.z + worldZ
        );
    }

    updateCharacterAnimation(delta, moving, running);

    // Gravity and jumping

    velocityY -= SETTINGS.gravity * delta;
    player.position.y += velocityY * delta;

    if (player.position.y <= 0) {
        player.position.y = 0;
        velocityY = 0;
        grounded = true;
    }
}

// ============================================================
// ENVIRONMENT UPDATE
// ============================================================

function updateEnvironment(delta) {
    for (const cloud of clouds) {
        cloud.object.position.x += cloud.speed * delta;

        if (cloud.object.position.x > SETTINGS.worldHalfSize) {
            cloud.object.position.x = -SETTINGS.worldHalfSize;
        }
    }
}

    if (!SETTINGS.debug) return;

    debugTimer += delta;

    if (debugTimer < 2) return;

    debugTimer = 0;

    console.log("===== GCA PERFORMANCE =====");
    console.log("Draw calls:", renderer.info.render.calls);
    console.log("Triangles:", renderer.info.render.triangles);
    console.log("Geometries:", renderer.info.memory.geometries);
    console.log("Textures:", renderer.info.memory.textures);
    console.log("Loaded trees:", interactiveTrees.length);
    console.log("Planned tree locations:", treePositionList.length);
    console.log(
        "World footprint:",
        SETTINGS.worldHalfSize * 2,
        "x",
        SETTINGS.worldHalfSize * 2
    );
    console.log("===========================");
}

    requestAnimationFrame(animate);

    updatePlayer(delta);
    updateTreeInteractions(delta);
    updateCamera();
    updateEnvironment(delta);

    updatePerformance(delta);
}
